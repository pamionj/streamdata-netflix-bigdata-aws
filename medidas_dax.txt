// =============================================================================
// 1. MÉTRICAS GENERALES DE VOLUMEN Y CATÁLOGO (TARJETAS KPI)
// =============================================================================

// 1.1 Total de Títulos Únicos en el Catálogo
Total Títulos = 
DISTINCTCOUNT(netflix_titles_clean[show_id])

// 1.2 Total de Películas (Movies)
Total Películas = 
CALCULATE(
    [Total Títulos],
    netflix_titles_clean[content_type] = "Movie"
)

// 1.3 Total de Series de TV (TV Shows)
Total Series = 
CALCULATE(
    [Total Títulos],
    netflix_titles_clean[content_type] = "TV Show"
)

// 1.4 Proporción de Películas (%)
Porcentaje Películas = 
DIVIDE([Total Películas], [Total Títulos], 0)

// 1.5 Proporción de Series (%)
Porcentaje Series = 
DIVIDE([Total Series], [Total Títulos], 0)


// =============================================================================
// 2. MÉTRICAS DE DIVERSIDAD GEOGRÁFICA Y CATEGORÍAS (TABLAS PUENTE)
// =============================================================================

// 2.1 Total de Países Únicos Participantes en Producción
Total Países = 
DISTINCTCOUNT(netflix_title_countries[country_name])

// 2.2 Total de Géneros / Categorías Únicas
Total Géneros = 
DISTINCTCOUNT(netflix_title_genres[genre])

// 2.3 Títulos Únicos Asociados a un País (Uso en gráficos de barras de Países)
Títulos por País = 
DISTINCTCOUNT(netflix_title_countries[show_id])

// 2.4 Títulos Únicos Asociados a un Género (Uso en gráficos de barras de Géneros)
Títulos por Género = 
DISTINCTCOUNT(netflix_title_genres[show_id])

// 2.5 Títulos con Coproducción Internacional (Multicountry)
Títulos Multicountry = 
CALCULATE(
    [Total Títulos],
    netflix_titles_clean[country_count] > 1
)

// 2.6 Porcentaje de Coproducción Multicountry (%)
Porcentaje Multicountry = 
DIVIDE(
    [Títulos Multicountry],
    CALCULATE(
        [Total Títulos],
        netflix_titles_clean[primary_country] <> "No informado"
    ),
    0
)


// =============================================================================
// 3. MÉTRICAS DE SENTIMIENTO Y TONO LÉXICO (NLP / TEXTBLOB)
// =============================================================================

// 3.1 Proporción de Sinopsis con Tono Positivo (%)
Porcentaje Tono Positivo = 
DIVIDE(
    CALCULATE(
        [Total Títulos],
        netflix_titles_clean[sentiment_label] = "Positivo"
    ),
    [Total Títulos],
    0
)

// 3.2 Proporción de Sinopsis con Tono Neutro (%)
Porcentaje Tono Neutro = 
DIVIDE(
    CALCULATE(
        [Total Títulos],
        netflix_titles_clean[sentiment_label] = "Neutro"
    ),
    [Total Títulos],
    0
)

// 3.3 Proporción de Sinopsis con Tono Negativo (%)
Porcentaje Tono Negativo = 
DIVIDE(
    CALCULATE(
        [Total Títulos],
        netflix_titles_clean[sentiment_label] = "Negativo"
    ),
    [Total Títulos],
    0
)


// =============================================================================
// 4. MÉTRICAS TEMPORALES Y DURACIÓN
// =============================================================================

// 4.1 Duración Promedio de Películas (en minutos)
Duración Promedio Películas = 
CALCULATE(
    AVERAGE(netflix_titles_clean[duration_value]),
    netflix_titles_clean[duration_unit] = "min"
)

// 4.2 Promedio de Temporadas en Series de TV
Promedio Temporadas Series = 
CALCULATE(
    AVERAGE(netflix_titles_clean[duration_value]),
    netflix_titles_clean[duration_unit] = "season"
)

// 4.3 Títulos de Incorporación Reciente (Años 2019 a 2021)
Títulos Recientes = 
CALCULATE(
    [Total Títulos],
    netflix_titles_clean[added_year] IN {2019, 2020, 2021}
)

// 4.4 Índice de Renovación de Catálogo (%)
Porcentaje Incorporación Reciente = 
DIVIDE(
    [Títulos Recientes],
    CALCULATE(
        [Total Títulos],
        NOT(ISBLANK(netflix_titles_clean[added_year]))
    ),
    0
)
